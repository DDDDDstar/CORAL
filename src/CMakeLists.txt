set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)
set(CMAKE_CUDA_SEPARABLE_COMPILATION ON)  # 开启可分离编译

file(GLOB_RECURSE CPP_SOURCES *.cpp)
file(GLOB_RECURSE CUDA_SOURCES *.cu)

find_package(CUDA REQUIRED)
include_directories(${CUDA_INCLUDE_DIRS})
link_directories(${CUDA_LIBRARY_DIRS}) 

# 查找BLAS库
find_package(BLAS REQUIRED)
message(STATUS "BLAS libraries: ${BLAS_LIBRARIES}")

add_library(${PROJECT_NAME} ${CPP_SOURCES} ${CUDA_SOURCES})
# 设置 CUDA 架构，根据实际 GPU 设置 sm_75 等
set(CMAKE_CUDA_ARCHITECTURES "80")
set_target_properties(${PROJECT_NAME} PROPERTIES
  CUDA_ARCHITECTURES "80"
  POSITION_INDEPENDENT_CODE ON
)
target_link_libraries(${PROJECT_NAME} ${BLAS_LIBRARIES} ${CUDA_LIBRARIES})

add_library(${PROJECT_NAME}_s STATIC ${CPP_SOURCES} ${CUDA_SOURCES})
# 设置 CUDA 架构，根据实际 GPU 设置 sm_75 等
set_target_properties(${PROJECT_NAME}_s PROPERTIES
  CUDA_ARCHITECTURES "75;80;86"
  POSITION_INDEPENDENT_CODE ON
)
target_link_libraries(${PROJECT_NAME}_s ${BLAS_LIBRARIES} ${CUDA_LIBRARIES})
